cmake_minimum_required(VERSION 3.25)
project(ParallelMaterComponentConsumer LANGUAGES C CXX)

set(PM_COMPONENTS loader CACHE STRING "Parallel Mater components to require")
find_package(ParallelMater 0.3 CONFIG REQUIRED COMPONENTS ${PM_COMPONENTS})

if("loader" IN_LIST PM_COMPONENTS)
    add_executable(loader-consumer devices.c)
    target_compile_features(loader-consumer PRIVATE c_std_11)
    target_link_libraries(loader-consumer PRIVATE ParallelMater::api)
endif()
if("validation" IN_LIST PM_COMPONENTS AND
   NOT TARGET ParallelMater::validation)
    message(FATAL_ERROR "validation component has no imported target")
endif()
if("preparation" IN_LIST PM_COMPONENTS)
    add_executable(preparation-consumer preparation_consumer.cpp)
    target_compile_features(preparation-consumer PRIVATE cxx_std_20)
    target_link_libraries(preparation-consumer PRIVATE ParallelMater::preparation)
endif()
if("scene" IN_LIST PM_COMPONENTS)
    add_executable(scene-consumer scene_consumer.cpp)
    target_compile_features(scene-consumer PRIVATE cxx_std_20)
    target_link_libraries(scene-consumer PRIVATE ParallelMater::scene)
endif()
if("driver" IN_LIST PM_COMPONENTS AND NOT ParallelMater_DRIVER_TARGETS)
    message(FATAL_ERROR "driver component has no imported driver targets")
endif()
foreach(expected IN LISTS PM_EXPECT_DRIVER_TARGETS)
    if(NOT TARGET "${expected}")
        message(FATAL_ERROR "driver component is missing ${expected}")
    endif()
endforeach()
if("conformance" IN_LIST PM_COMPONENTS)
    if(NOT EXISTS "${ParallelMater_CONFORMANCE_EXECUTABLE}" OR
       NOT IS_DIRECTORY "${ParallelMater_CONFORMANCE_DATA_ROOT}")
        message(FATAL_ERROR "conformance component paths are incomplete")
    endif()
endif()
if("benchmark" IN_LIST PM_COMPONENTS AND
   NOT EXISTS "${ParallelMater_BENCHMARK_QUALIFICATION}")
    message(FATAL_ERROR "benchmark qualification tool is missing")
endif()
